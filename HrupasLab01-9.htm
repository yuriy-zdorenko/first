Програмування

Нижче наведено приклад коду мовою Python, який демонструє роботу зі стандартним виводом та обчисленням математичного виразу: 

def calculate_sum(a, b):
    # Функція повертає суму двох чисел
    result = a + b
    return result

def main():
    x = 10
    y = 25
    total = calculate_sum(x, y)
    print(f"Результат обчислення: {total}")

if __name__ == "__main__":
    main()
